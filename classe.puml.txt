@startuml
' Classes do Sistema

enum Setor {
    ALIMENTACAO
    TECNOLOGIA
    VAREJO
    EDUCACAO
    SAUDE
    TRANSPORTE
    ENTRETENIMENTO
    SERVICOS
}

class Usuario {
    -email: String
    -senha: String
    --
    +Autenticar(): Boolean
    +AlterarSenha(): Void
}

class Instituicao {
    nome: String
    cnpj: String
    endereço: String
}

class Aluno {
    -nome: String
    -cpf: String
    -rg: String
    -endereco: String
    -curso: String
    -saldoMoedas: Integer
    --
    +cadastrar(): void
    +consultarExtrato(): List<Transacao>
    +trocarMoedas(vantagem: Vantagem): Cupom
}

class Professor {
    -nome: String
    -cpf: String
    -departamento: String
    --
    +cadastrar(): void
    +consultarExtrato(): List<Transacao>
}

class EmpresaParceira {
    -razãoSocial: String
    -cnpj: String
    -setor: Setor
    --
    +cadastrar(): void
}

class Transacao {
    -valor: Double
    -data: String
    -descricao: String
    --
}

class TransacaoEnvio {
   -alunoBeneficiado: Aluno
    --
    +enviar(): void
}

class TransacaoCompra {
    #{field} vantagem: Vantagem
    --
    +comprar(): void
}

class Cupom {
    -{field} codigo: String
    -{field} dataEmissao: String
    -{field} dataValidade: String
    --
    +usar(): Boolean
}

class Vantagem {
    -descricao: String
    -preco: Integer
    --
    +descrever(): String
}

' Relacionamentos

Usuario <|-- Aluno
Usuario <|-- Professor

Instituicao "1" o-- "*" Aluno : vincula-se
Instituicao "1" o-- "*" Professor : faz parte

Professor "1" --> "*" TransacaoEnvio : realiza
TransacaoEnvio "1" --> "*" Aluno : recebe

Aluno "1" --> "*" Cupom : resgata

Transacao <|-- TransacaoEnvio
Transacao <|-- TransacaoCompra
Transacao <|-- Cupom

EmpresaParceira "1" *-- "*" Vantagem : oferece

Cupom "*" --> "1" Vantagem : referente a
@enduml
